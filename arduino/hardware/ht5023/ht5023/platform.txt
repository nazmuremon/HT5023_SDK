compiler.c.elf.flags=-mcpu=cortex-m0 -mthumb {compiler.optimization_flags} {compiler.warning_flags} -fno-exceptions -fno-rtti -fno-threadsafe-statics
compiler.S.flags=-c -mcpu=cortex-m0 -mthumb -x assembler-with-cpp

recipe.S.o.pattern="{compiler.path}{compiler.S.cmd}" {compiler.S.flags} {compiler.define.flags} {compiler.S.extra_flags} {build.extra_flags} -DF_CPU={build.f_cpu} -DARDUINO={runtime.ide.version} -DARDUINO_{build.board} {includes} "{source_file}" -o "{object_file}"

pluggable_discovery.required.0=builtin:serial-discovery

# Upload through the J-Link (Standalone layout: the sketch owns 0x0000).
#
# ".linux"/".macosx" suffixes override the plain key on those systems. They are
# needed because "python" is a Windows-only spelling: most Linux distributions
# ship only "python3" and have no "python" at all, so an unqualified {cmd}
# fails with "command not found" the moment you press Upload.
tools.ht5023jlink.cmd=python
tools.ht5023jlink.cmd.linux=python3
tools.ht5023jlink.cmd.macosx=python3
tools.ht5023jlink.upload.params.verbose=
tools.ht5023jlink.upload.params.quiet=
tools.ht5023jlink.upload.pattern="{cmd}" "{runtime.platform.path}/tools/ht5023_upload.py" --method jlink --speed {upload.speed} --bin "{build.path}/{build.project_name}.bin"
tools.ht5023jlink.program.pattern="{cmd}" "{runtime.platform.path}/tools/ht5023_upload.py" --method jlink --speed {upload.speed} --bin "{build.path}/{build.project_name}.bin"

# Upload over USB-UART through the resident bootloader (UART bootloader layout:
# the sketch is linked at 0x2000). Uses the port selected in the IDE, and the
# board has to be reset while it is listening.
tools.ht5023usb.cmd=python
tools.ht5023usb.cmd.linux=python3
tools.ht5023usb.cmd.macosx=python3
tools.ht5023usb.upload.params.verbose=
tools.ht5023usb.upload.params.quiet=
tools.ht5023usb.upload.pattern="{cmd}" "{runtime.platform.path}/tools/ht5023_upload.py" --method usb --port "{serial.port}" --baud {upload.speed} --bin "{build.path}/{build.project_name}.bin"
tools.ht5023usb.program.pattern="{cmd}" "{runtime.platform.path}/tools/ht5023_upload.py" --method usb --port "{serial.port}" --baud {upload.speed} --bin "{build.path}/{build.project_name}.bin"
